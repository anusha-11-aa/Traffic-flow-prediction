import pandas as pd
import numpy as np
import matplotlib.pyplot as plt
from sklearn.preprocessing import MinMaxScaler
from sklearn.metrics import mean_absolute_error, mean_squared_error, r2_score
from tensorflow.keras.models import Sequential
from tensorflow.keras.layers import LSTM, Dense, Dropout
from tensorflow.keras.callbacks import EarlyStopping, ReduceLROnPlateau

# Load dataset
data = pd.read_csv("traffic.csv")

# Rename
data.rename(columns={
    'CarCount': 'Cars',
    'BikeCount': 'Bikes',
    'BusCount': 'Buses',
    'TruckCount': 'Trucks'
}, inplace=True)

# Total vehicles
data['Total_Vehicles'] = data['Cars'] + data['Bikes'] + data['Buses'] + data['Trucks']

# Datetime
data['Datetime'] = pd.to_datetime(
    '2023-01-' + data['Date'].astype(str) + ' ' + data['Time'],
    format='%Y-%m-%d %I:%M:%S %p'
)

data = data.sort_values(by='Datetime')

# 🔥 NEW FEATURE ENGINEERING (IMPORTANT FOR ACCURACY)
data['Hour'] = data['Datetime'].dt.hour
data['Day'] = data['Datetime'].dt.day
data['Weekday'] = data['Datetime'].dt.weekday

# Cyclical encoding (VERY IMPORTANT)
data['Hour_sin'] = np.sin(2 * np.pi * data['Hour']/24)
data['Hour_cos'] = np.cos(2 * np.pi * data['Hour']/24)

# Encode traffic situation
if 'Traffic Situation' in data.columns:
    data['Traffic Situation'] = data['Traffic Situation'].astype('category').cat.codes

# Use important features
features = [
    'Cars', 'Bikes', 'Buses', 'Trucks',
    'Hour_sin', 'Hour_cos', 'Weekday',
    'Total_Vehicles'
]

df = data[features]

# Scaling
scaler = MinMaxScaler()
scaled_data = scaler.fit_transform(df)

# Sequence creation
def create_sequences(data, seq_length):
    X, y = [], []
    for i in range(len(data) - seq_length):
        X.append(data[i:i + seq_length, :-1])
        y.append(data[i + seq_length, -1])
    return np.array(X), np.array(y)

seq_length = 15   # 🔥 balanced (better accuracy)
X, y = create_sequences(scaled_data, seq_length)

# Split
split = int(0.8 * len(X))
X_train, X_test = X[:split], X[split:]
y_train, y_test = y[:split], y[split:]

# 🔥 BETTER MODEL (not too big, not too small)
model = Sequential([
    LSTM(100, return_sequences=True, input_shape=(X_train.shape[1], X_train.shape[2])),
    Dropout(0.2),
    LSTM(50),
    Dense(25, activation='relu'),
    Dense(1)
])

model.compile(optimizer='adam', loss='mse')  # 🔥 MSE improves regression accuracy

model.summary()

# Callbacks
early_stop = EarlyStopping(monitor='val_loss', patience=8, restore_best_weights=True)
reduce_lr = ReduceLROnPlateau(monitor='val_loss', factor=0.3, patience=4)

# Train
history = model.fit(
    X_train, y_train,
    epochs=80,
    batch_size=32,
    validation_data=(X_test, y_test),
    callbacks=[early_stop, reduce_lr],
    verbose=1
)

# Predict
y_pred = model.predict(X_test)

# Inverse scaling
dummy = np.zeros((X_test.shape[0], df.shape[1]))

dummy[:, -1] = y_test
inv_y_test = scaler.inverse_transform(dummy)[:, -1]

dummy[:, -1] = y_pred.flatten()
inv_y_pred = scaler.inverse_transform(dummy)[:, -1]

# Metrics
mae = mean_absolute_error(inv_y_test, inv_y_pred)
rmse = np.sqrt(mean_squared_error(inv_y_test, inv_y_pred))
r2 = r2_score(inv_y_test, inv_y_pred)

print("\n📊 High Accuracy Model Metrics:")
print(f"MAE: {mae:.2f}")
print(f"RMSE: {rmse:.2f}")
print(f"R² Score: {r2:.4f}")

# Plot
plt.figure(figsize=(10, 6))
plt.plot(inv_y_test, label='Actual')
plt.plot(inv_y_pred, label='Predicted')
plt.legend()
plt.title("🚗 High Accuracy Traffic Prediction")
plt.show()

print(f"\n✅ Accuracy: {r2*100:.2f}%")
