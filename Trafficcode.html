import pandas as pd
import matplotlib.pyplot as plt
import seaborn as sns
import numpy as np
# ==========================
# 🔹 Load & Preprocess Data
# ==========================
file_name = 'traffic flow prediction.csv'
df = pd.read_csv(file_name)

# 1. Drop "Day of the week" column
df = df.drop(columns=["Day of the week"])

# 2. Convert categorical column "Traffic Situation" into numeric codes
df["Traffic Situation"] = df["Traffic Situation"].astype("category").cat.codes

# 3. Convert Date + Time into proper DateTime format (assuming Jan 2025)
df["Date"] = df["Date"].astype(int)
df["DateTime"] = pd.to_datetime(
    "2025-01-" + df["Date"].astype(str) + " " + df["Time"],
    format="%Y-%m-%d %I:%M:%S %p"
)

# 4. Drop old Date and Time columns
df = df.drop(columns=["Date", "Time"])

# 5. Reorder columns (DateTime first)
df = df[["DateTime", "CarCount", "BikeCount", "BusCount", "TruckCount", "Traffic Situation"]]

# 6. Add Total Vehicles column
df["Total"] = df["CarCount"] + df["BikeCount"] + df["BusCount"] + df["TruckCount"]

print("✅ First 5 rows of the preprocessed dataset:")
print(df.head())
print("\n✅ Dataset shape:", df.shape)


# ==========================
# 🔹 Visualizations
# ==========================

# Correlation heatmap of numerical features
plt.figure(figsize=(10, 6))
sns.heatmap(df[['CarCount','BikeCount','BusCount','TruckCount','Total']].corr(), 
            cmap='coolwarm', annot=True, linewidths=0.5)
plt.title('Feature Correlation Heatmap')
plt.show()

# Boxplot of Total traffic by Traffic Situation
plt.figure(figsize=(8, 5))
sns.boxplot(x='Traffic Situation', y='Total', data=df)
plt.title('Traffic Volume by Traffic Situation')
plt.xlabel('Traffic Situation')
plt.ylabel('Total Vehicles')
plt.show()

# Line plot of traffic over time
plt.figure(figsize=(12, 6))
plt.plot(df["DateTime"], df["Total"], color="blue", linewidth=1.5)
plt.title("Traffic Volume Over Time")
plt.xlabel("DateTime")
plt.ylabel("Total Vehicles")
plt.xticks(rotation=45)
plt.tight_layout()
plt.show()
